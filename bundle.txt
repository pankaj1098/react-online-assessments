/** One sidebar row. If the module has sub modules it gets an expand button and nests them. */
function SidebarItem({ module, collapsed = false, depth = 0, onNavigate }) {
  const ability = useAbility()
  const { pathname } = useLocation()
  // null = follow the URL (open while you are inside this branch); true/false = user's choice.
  const [toggled, setToggled] = useState(null)
  const children = (module.children ?? []).filter((child) => ability.can('view', child.id))
  const hasChildren = children.length > 0 && !collapsed
  const inBranch = findModuleTrail(pathname).some((item) => item.id === module.id)
  const expanded = hasChildren && (toggled ?? inBranch)
  const Icon = module.icon
  return (
    <li>
      <div className="relative">
        <NavLink
          to={module.path}
          // Parents match their own page only, so the highlight moves to the sub module.
          end={module.path === '/' || children.length > 0}
          onClick={onNavigate}
          title={module.name}
          className={({ isActive }) =>
            cn(
              'flex items-center rounded-xl transition-colors',
              depth === 0 ? 'min-h-12 gap-4 text-base' : 'min-h-10 gap-3 text-sm',
              collapsed ? 'justify-center' : depth === 0 ? 'px-4' : 'px-3',
              hasChildren && 'pr-12',
              isActive
                ? 'bg-sidebar-active font-bold text-sidebar-active-foreground shadow-sm'
                : cn('text-sidebar-foreground hover:bg-sidebar-hover', inBranch && 'font-bold'),
            )
          }
        >
          {Icon && <Icon className="size-5.5 shrink-0" aria-hidden="true" />}
          <span className={cn(collapsed && 'sr-only')}>{module.name}</span>
        </NavLink>
        {hasChildren && (
          <button
            type="button"
            aria-expanded={expanded}
            aria-label={`${expanded ? 'Collapse' : 'Expand'} ${module.name}`}
            onClick={() => setToggled(!expanded)}
            className="absolute top-1/2 right-2 -translate-y-1/2 rounded-lg p-2 hover:bg-sidebar-hover"
          >
            <ChevronDown
              className={cn('size-4 transition-transform', expanded && 'rotate-180')}
              aria-hidden="true"
            />
          </button>
        )}
      </div>
      {expanded && (
        <ul className="mt-1 mb-1 ml-9 grid gap-1 border-l border-sidebar-border pl-2">
          {children.map((child) => (
            <SidebarItem key={child.id} module={child} depth={depth + 1} onNavigate={onNavigate} />
          ))}
        </ul>
      )}
    </li>
  )
}

/** All modules at every level, in order. */
export function flattenModules(list = modules) {
  return list.flatMap((module) => [module, ...flattenModules(module.children ?? [])])
}

/** Path from the top-level module down to the one matching the URL, e.g. [Admin, Gate Pass]. */
export function findModuleTrail(pathname, list = modules, trail = []) {
  let best = []
  for (const module of list) {
    if (module.path === '/') continue
    const next = [...trail, module]
    if (pathname === module.path || pathname.startsWith(`${module.path}/`)) best = next
    const deeper = findModuleTrail(pathname, module.children ?? [], next)
    if (deeper.length > best.length) best = deeper
  }
  return best
}

export function findModuleByPath(pathname) {
  return findModuleTrail(pathname).at(-1)
}

import { z } from 'zod'
import { daysBetween, todayLocal } from '../format'

export const REQUEST_MAX_LENGTH = 500
export const MAX_BOOKING_DAYS = 31

/** `endDate` is optional: blank means a single day. */
export const bookingFormSchema = z
  .object({
    title: z
      .string()
      .trim()
      .min(1, 'Enter the meeting title.')
      .max(150, 'Title is too long (max 150 characters).'),
    persons: z
      .string()
      .trim()
      .regex(/^[1-9]\d{0,2}$/, 'Enter the number of persons (1-999).'),
    additionalRequest: z
      .string()
      .trim()
      .max(REQUEST_MAX_LENGTH, `Request is too long (max ${REQUEST_MAX_LENGTH} characters).`),
    startDate: z.string().min(1, 'Select the start date.'),
    endDate: z.string(),
    startTime: z.string().min(1, 'Select the start time.'),
    endTime: z.string().min(1, 'Select the end time.'),
    roomId: z.string().min(1, 'Select a room.'),
  })
  .superRefine((values, ctx) => {
    const end = values.endDate || values.startDate
    if (values.startDate && values.startDate < todayLocal()) {
      ctx.addIssue({
        code: 'custom',
        path: ['startDate'],
        message: 'The start date cannot be in the past.',
      })
    }
    if (values.startDate && end < values.startDate) {
      ctx.addIssue({
        code: 'custom',
        path: ['endDate'],
        message: 'End date must be on or after the start date.',
      })
    } else if (values.startDate && daysBetween(values.startDate, end) > MAX_BOOKING_DAYS) {
      ctx.addIssue({
        code: 'custom',
        path: ['endDate'],
        message: `A request can cover at most ${MAX_BOOKING_DAYS} days.`,
      })
    }
    if (values.startTime && values.endTime && values.endTime <= values.startTime) {
      ctx.addIssue({
        code: 'custom',
        path: ['endTime'],
        message: 'End time must be after the start time.',
      })
    }

/** Date range a calendar view needs for the given anchor day. */
export function rangeFor(view, anchor) {
  if (view === 'month') return monthGridRange(anchor)
  if (view === 'week') {
    const from = startOfWeek(anchor)
    return { from, to: addDays(from, 6) }
  }
  if (view === 'day') return { from: anchor, to: anchor }
  return { from: startOfMonth(anchor), to: endOfMonth(anchor) }
}

/** Move the anchor one step (direction -1 or 1) for the given view. */
export function shift(view, anchor, direction) {
  if (view === 'week') return addDays(anchor, 7 * direction)
  if (view === 'day') return addDays(anchor, direction)
  return shiftMonth(anchor, direction)
}

export function titleFor(view, anchor) {
  if (view === 'week') {
    const from = startOfWeek(anchor)
    return `${formatDayMonth(from)} - ${formatDayMonth(addDays(from, 6))}`
  }
  if (view === 'day') return formatDateLong(anchor)
  return formatMonthYear(anchor)
}

  const [roomId, startDate, endDate, startTime, endTime, additionalRequest] = useWatch({
    control,
    name: ['roomId', 'startDate', 'endDate', 'startTime', 'endTime', 'additionalRequest'],
  })
  const room = rooms.find((item) => item.id === roomId)
  const effectiveEnd = isEdit ? startDate : endDate || startDate

  // Check availability once everything needed is filled in and valid.
  const ready = Boolean(
    roomId &&
    startDate &&
    startTime &&
    endTime &&
    startDate >= todayLocal() &&
    effectiveEnd >= startDate &&
    daysBetween(startDate, effectiveEnd) <= MAX_BOOKING_DAYS &&
    endTime > startTime,

  const slot = ready ? [roomId, startDate, effectiveEnd, startTime, endTime].join('|') : ''

function groupByDate(events) {
  const groups = new Map()
  for (const event of events) {
    const list = groups.get(event.date) ?? []
    list.push(event)
    groups.set(event.date, list)
  }
  for (const list of groups.values()) list.sort((a, b) => a.startTime.localeCompare(b.startTime))
  return groups
}

function EventRow({ event, toneOf }) {
  return (
    <li className={cn('rounded-lg px-3 py-2.5 text-sm', toneStyles(toneOf(event.roomId)).event)}>
      <p className="font-semibold">{formatTimeRange(event.startTime, event.endTime)}</p>
      <p className="mt-0.5 break-words">{event.title}</p>
      <p className="text-muted-foreground">
        {event.roomName} &middot; {event.requesterName}
        {event.department ? `, ${event.department}` : ''}
      </p>
    </li>
  )
}

function MonthView({ anchor, events, toneOf, onPickDay }) {
  const { from, to } = monthGridRange(anchor)
  const byDate = groupByDate(events)
  const month = anchor.slice(0, 7)
  const today = todayLocal()
  return (
    <div className="overflow-x-auto">
      <div className="grid min-w-[42rem] grid-cols-7 gap-px overflow-hidden rounded-xl border border-border bg-border">
        {weekdayLabels.map((label) => (
          <div
            key={label}
            className="bg-muted px-2 py-2 text-center text-xs font-bold text-muted-foreground"
          >
            {label}
          </div>
        ))}
        {eachDay(from, to).map((date) => {
          const dayEvents = byDate.get(date) ?? []
          const inMonth = date.slice(0, 7) === month
          return (
            <div key={date} className={cn('min-h-28 bg-card p-1.5', !inMonth && 'bg-muted/50')}>
              <button
                type="button"
                aria-label={`Open ${formatDateLong(date)}`}
                onClick={() => onPickDay(date)}
                className={cn(
                  'mb-1 grid size-7 place-items-center rounded-full text-sm font-semibold hover:bg-secondary',
                  !inMonth && 'text-muted-foreground',
                  date === today && 'bg-primary text-primary-foreground hover:bg-primary-hover',
                )}
              >
                {Number(date.slice(8))}
              </button>
              <ul className="grid gap-1">
                {dayEvents.slice(0, 3).map((event) => (

                </button>

            </div>

/** Month, week, day and list views over the same list of events. */
export function BookingsCalendar({ view, anchor, events, toneOf, onPickDay }) {
  const byDate = groupByDate(events)
  if (view === 'month') {
    return <MonthView anchor={anchor} events={events} toneOf={toneOf} onPickDay={onPickDay} />
  }
  if (view === 'week') {
    const first = startOfWeek(anchor)
    return (
      <div className="grid gap-3 md:grid-cols-2 xl:grid-cols-7">
        {Array.from({ length: 7 }, (_, index) => addDays(first, index)).map((date) => (
          <DayColumn
            key={date}
            date={date}
            events={byDate.get(date) ?? []}
            toneOf={toneOf}
            header="short"
          />
        ))}
      </div>
    )
  }
  if (view === 'day') {
    return (
      <DayColumn date={anchor} events={byDate.get(anchor) ?? []} toneOf={toneOf} header="none" />
    )
  }
  const dates = [...byDate.keys()].sort()
  if (dates.length === 0) return <EmptyState>No approved bookings in this period.</EmptyState>
  return (
    <div className="grid gap-3">
      {dates.map((date) => (
        <DayColumn key={date} date={date} events={byDate.get(date)} toneOf={toneOf} />
      ))}
    </div>
  )
}

const pad = (value) => String(value).padStart(2, '0')

export const weekdayLabels = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun']
export const weekdayNames = [
  'Monday',
  'Tuesday',
  'Wednesday',
  'Thursday',
  'Friday',
  'Saturday',
  'Sunday',
]

/** Local calendar date as "YYYY-MM-DD" (not UTC). */
export function toDateString(date) {
  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`
}

export function todayLocal() {
  return toDateString(new Date())
}

export function toLocalDate(dateString) {
  return new Date(`${dateString}T00:00`)
}

export function addDays(dateString, days) {
  const date = toLocalDate(dateString)
  date.setDate(date.getDate() + days)
  return toDateString(date)
}

export function shiftMonth(dateString, months) {
  const date = toLocalDate(dateString)
  date.setDate(1)
  date.setMonth(date.getMonth() + months)
  return toDateString(date)
}

export function startOfMonth(dateString) {
  return `${dateString.slice(0, 7)}-01`
}

export function endOfMonth(dateString) {
  const date = toLocalDate(dateString)
  return toDateString(new Date(date.getFullYear(), date.getMonth() + 1, 0))
}

/** Monday = 0 ... Sunday = 6. */
export function weekdayIndex(dateString) {
  return (toLocalDate(dateString).getDay() + 6) % 7
}

export function startOfWeek(dateString) {
  return addDays(dateString, -weekdayIndex(dateString))
}

/** Number of days from start to end, both included. */
export function daysBetween(start, end) {
  return Math.round((toLocalDate(end) - toLocalDate(start)) / 86_400_000) + 1
}

export function eachDay(start, end) {
  return Array.from({ length: daysBetween(start, end) }, (_, index) => addDays(start, index))
}

/** The full Monday-to-Sunday grid that covers a month. */
export function monthGridRange(dateString) {
  return {
    from: startOfWeek(startOfMonth(dateString)),
    to: addDays(startOfWeek(endOfMonth(dateString)), 6),
  }
}

export function dateParts(dateString) {
  const date = toLocalDate(dateString)
  return { day: date.getDate(), month: date.toLocaleString('en-GB', { month: 'short' }) }
}

/** "14:30" -> "2:30 PM". */
export function formatTime(value) {
  const [hours, minutes] = value.split(':').map(Number)
  return `${hours % 12 || 12}:${pad(minutes)} ${hours < 12 ? 'AM' : 'PM'}`
}

export function formatTimeRange(start, end) {
  return `${formatTime(start)} - ${formatTime(end)}`
}

export function toMinutes(value) {
  const [hours, minutes] = value.split(':').map(Number)
  return hours * 60 + minutes
}

export const formatHourShort = (hour) => `${hour % 12 || 12}${hour < 12 ? 'a' : 'p'}`

export const formatHourLong = (hour) => `${hour % 12 || 12} ${hour < 12 ? 'am' : 'pm'}`

export function formatDateLong(dateString) {
  return toLocalDate(dateString).toLocaleDateString('en-GB', {
    weekday: 'long',
    day: 'numeric',
    month: 'long',
    year: 'numeric',
  })
}

export function formatDayMonth(dateString) {
  return toLocalDate(dateString).toLocaleDateString('en-GB', { day: 'numeric', month: 'short' })
}

export function formatMonthYear(dateString) {
  return toLocalDate(dateString).toLocaleDateString('en-GB', { month: 'long', year: 'numeric' })
}

/** "Mon 23" - short column header for the week view. */
export function formatWeekdayDay(dateString) {
  return `${weekdayLabels[weekdayIndex(dateString)]} ${Number(dateString.slice(8))}`
}

export const meetingRoomKeys = {
  all: ['meeting-rooms'],
  rooms: () => [...meetingRoomKeys.all, 'rooms'],
  list: (params) => [...meetingRoomKeys.all, 'list', params],
  detail: (id) => [...meetingRoomKeys.all, 'detail', id],
  calendar: (params) => [...meetingRoomKeys.all, 'calendar', params],
  insights: (params) => [...meetingRoomKeys.all, 'insights', params],
  availability: (params) => [...meetingRoomKeys.all, 'availability', params],
}

  return useQuery({
    queryKey: meetingRoomKeys.rooms(),
    queryFn: meetingRoomApi.rooms,
    staleTime: 5 * 60_000,
  })
}

  return useQuery({
    queryKey: meetingRoomKeys.list(params),
    queryFn: () => meetingRoomApi.list(params),
    placeholderData: keepPreviousData,
  })

const joinWords = (items) =>
  items.length <= 2 ? items.join(' and ') : `${items.slice(0, -1).join(', ')} and ${items.at(-1)}`

/** "Busiest at 9 AM and 3 PM. Rooms are easiest to get after 5 PM." */
function timingInsight(hourData) {
  const max = Math.max(0, ...hourData.map((item) => item.value))
  if (max === 0) return null
  const busiest = hourData
    .filter((item) => item.value === max)
    .map((item) => item.title.toUpperCase())
  const min = Math.min(...hourData.map((item) => item.value))
  const quietest = hourData.filter((item) => item.value === min)
  const trailing = quietest.at(-1) === hourData.at(-1)
  const easy = trailing
    ? `after ${(hourData[hourData.indexOf(quietest[0]) - 1] ?? quietest[0]).title.toUpperCase()}`
    : `around ${quietest[0].title.toUpperCase()}`
  return { busiest: joinWords(busiest), easy }
}

/** "Saturday is the busiest day. Wed to Fri have no bookings." */
function weekdayInsight(dayData) {
  const max = Math.max(0, ...dayData.map((item) => item.value))
  if (max === 0) return null
  const busiest = dayData
    .map((item, index) => (item.value === max ? weekdayNames[index] : null))
    .filter(Boolean)
  const quiet = []
  let run = []
  for (const [index, item] of dayData.entries()) {
    if (item.value === 0) run.push(index)
    else if (run.length) {
      quiet.push(run)
      run = []
    }
  }
  if (run.length) quiet.push(run)
  const phrases = quiet.map((days) =>
    days.length === 1
      ? weekdayLabels[days[0]]
      : `${weekdayLabels[days[0]]} to ${weekdayLabels[days.at(-1)]}`,
  )
  const quietDays = quiet.reduce((sum, days) => sum + days.length, 0)
  return {
    busiest: joinWords(busiest),
    plural: busiest.length > 1,
    quiet: phrases.length
      ? `${joinWords(phrases)} ${quietDays === 1 ? 'has' : 'have'} no bookings.`
      : '',
  }
}

/** isCancelled wins; then isApproved: true = Accepted, false = Rejected, null = Pending. */
export function getBookingStatus({ isCancelled, isApproved }) {
  if (isCancelled === true) return 'Cancelled'
  if (isApproved === true) return 'Accepted'
  if (isApproved === false) return 'Rejected'
  return 'Pending'
}

import { createMongoAbility } from '@casl/ability'
import { flattenModules } from '@/config/modules'

/** Modules that need a specific role. Admin is open to every employee, as in the old portal. */
const restrictedModules = {}

/**
 * Builds the permission set for a signed-in user.
 * Roles come from the .NET login response (`user.roles`).
 */
export function defineAbilityFor(roles) {
  // Not signed in: no permissions at all.
  if (!roles) return createMongoAbility([])
  const userRoles = roles
  const rules = []
  if (userRoles.includes('admin')) {
    rules.push({ action: 'manage', subject: 'all' })
  } else {
    for (const { id } of flattenModules()) {
      const required = restrictedModules[id]
      if (!required || userRoles.includes(required)) {
        rules.push({ action: 'view', subject: id })
      }
    }
    rules.push(
      { action: 'create', subject: 'gate-pass' },
      { action: 'update', subject: 'gate-pass' },
      { action: 'create', subject: 'meeting-room-bookings' },
      { action: 'update', subject: 'meeting-room-bookings' },
    )
  }
  return createMongoAbility(rules)
}
