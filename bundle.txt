/** 62 → '1 hour 2 minutes', 525600 → '365 days'. */
function describeMinutes(total: number): string {
  const days = Math.floor(total / 1440);
  const hours = Math.floor((total % 1440) / 60);
  const minutes = total % 60;
  const parts = [
    days ? `${formatCount(days)} ${days === 1 ? 'day' : 'days'}` : '',
    hours ? `${hours} ${hours === 1 ? 'hour' : 'hours'}` : '',
    minutes ? `${minutes} ${minutes === 1 ? 'minute' : 'minutes'}` : '',
  ].filter(Boolean);
  return parts.join(' ') || '0 minutes';
}

function ExpiryHint() {
  const form = Form.useFormInstance<EarningPointsFormInput>();
  const minutes: unknown = Form.useWatch('expiryMinutes', form);
  const readable = typeof minutes === 'number' && minutes > 0 ? ` That is ${describeMinutes(minutes)}.` : '';
  return <>Earned points expire after this many minutes. The wallet needs at least 61.{readable}</>;
}

/** The metal of a tier card, found from the tier name. Other names get the plain card. */
const TIER_METALS = new Map<string, string>([
  ['gold', 'tier-card--gold'],
  ['silver', 'tier-card--silver'],
  ['platinum', 'tier-card--platinum'],
]);

function getTierCardClass(name: unknown): string {
  const lowerName = typeof name === 'string' ? name.toLowerCase() : '';
  const metal = [...TIER_METALS].find(([word]) => lowerName.includes(word));
  return metal ? `tier-card ${metal[1]}` : 'tier-card';
}

/** The line at the top of a tier card: its name, where it starts, and whether it is the base tier. */
function TierHeader({ index }: { index: number }) {
  const form = Form.useFormInstance<EarningPointsFormInput>();
  const name: unknown = Form.useWatch(['tiers', index, 'name'], form);
  const minPoints: unknown = Form.useWatch(['tiers', index, 'minPoints'], form);
  const multiplier: unknown = Form.useWatch(['tiers', index, 'earnMultiplier'], form);
  return (
    <span className="tier-header">
      <span className="tier-header__badge">
        <CrownOutlined aria-hidden="true" />
      </span>
      <span className="tier-header__name">{typeof name === 'string' && name.trim() ? name : 'New tier'}</span>
      {typeof minPoints === 'number' && <span className="settings-muted">from {formatCount(minPoints)} points</span>}
      {typeof multiplier === 'number' && <span className="settings-muted">· {multiplier}× birthday points</span>}
      {minPoints === 0 && <Tag color="blue">Base tier</Tag>}
    </span>
  );
}

function TierList() {
  const form = Form.useFormInstance<EarningPointsFormInput>();
  const [open, setOpen] = useState<string[]>([]);
  // Watching the tiers keeps each card's color in step with the name being typed.
  const tiers: unknown = Form.useWatch('tiers', { form, preserve: true });
  const tierRows = Array.isArray(tiers) ? (tiers as { name?: unknown }[]) : [];

  return (
    <Form.List name="tiers">
      {(rows, { add, remove }, { errors }) => (
        <>
          <Collapse
            className="tier-list"
            activeKey={open}
            onChange={(keys) => setOpen([keys].flat().map(String))}
            items={rows.map((row) => ({
              key: String(row.name),
              className: getTierCardClass(tierRows.at(row.name)?.name),
              label: <TierHeader index={row.name} />,
              forceRender: true,
              extra: (
                <Popconfirm
                  title="Delete this tier?"
                  description="It is removed when you save. The server refuses this if customers are in the tier."
                  okText="Delete"
                  okButtonProps={{ danger: true }}
                  cancelText="Cancel"
                  onConfirm={() => {
                    setOpen([]);
                    remove(row.name);
                  }}
                  onPopupClick={(event) => event.stopPropagation()}
                >
                  <Button
                    size="small"
                    type="text"
                    danger
                    icon={<DeleteOutlined />}
                    onClick={(event) => event.stopPropagation()}
                    aria-label="Delete tier"
                  />
                </Popconfirm>
              ),
              children: <TierFields index={row.name} />,
            }))}
          />
          <Form.ErrorList errors={errors} />
          <Button
            className="edit-list__add"
            icon={<PlusOutlined />}
            onClick={() => {
              add({ earnMultiplier: 1, benefits: '', discount: {}, cashback: {}, freeItem: {} });
              setOpen([String(rows.length)]);
            }}
          >
            Add tier
          </Button>
        </>
      )}
    </Form.List>
  );
}
