import { CaretDownFilled, CaretUpFilled } from '@ant-design/icons';
import { Card } from 'antd';
import type { Change } from '@/features/dashboard/format';

type KpiCardProps = {
  label: string;
  /** Already formatted, for example '₹48,260' */
  value: string;
  /** Small text after the value, for example '· 7.9%' */
  valueNote?: string;
  change: Change;
  /** Is a rise good news? (More revenue: yes. More cancelled orders: no.) Decides the green or red color. */
  higherIsBetter: boolean;
  comparedTo: string;
};

/** One number with its change against the previous period. */
export function KpiCard({ label, value, valueNote, change, higherIsBetter, comparedTo }: KpiCardProps) {
  const isGood = change.direction === 'flat' ? undefined : (change.direction === 'up') === higherIsBetter;
  const tone = isGood === undefined ? 'neutral' : isGood ? 'positive' : 'negative';
  const directionWord = change.direction === 'up' ? 'Up' : change.direction === 'down' ? 'Down' : 'No change';

  return (
    <Card className="dash-card dash-kpi" variant="borderless">
      <div className="dash-kpi__label">{label}</div>
      <div className="dash-kpi__value">
        {value}
        {valueNote && <span className="dash-kpi__note">{valueNote}</span>}
      </div>
      <div className="dash-kpi__change">
        <span className={`dash-chip dash-tone--${tone}`}>
          {change.direction === 'up' && <CaretUpFilled aria-hidden="true" />}
          {change.direction === 'down' && <CaretDownFilled aria-hidden="true" />}
          <span className="visually-hidden">{directionWord} </span>
          {change.text}
        </span>
        <span className="dash-kpi__compared">{comparedTo}</span>
      </div>
    </Card>
  );
}

import dayjs from 'dayjs';
import type { DailyStat } from '@/features/reports/types';

/**
 * SAMPLE DATA: daily order totals for the last ~2 years, ending today.
 * Repeatable (seeded), with a gentle growth trend, busier weekends and occasional spikes,
 * so the tiles, chart and downloads behave like real data.
 *
 * TODO: Replace with a reports API call (e.g. GET /reports/daily?from=&to=), then delete this file.
 */

const DAY_COUNT = 800;

/** Small seeded random generator (mulberry32): same numbers on every load. */
function createRandom(seed: number) {
  let state = seed;
  return () => {
    state = (state + 0x6d2b79f5) | 0;
    let t = Math.imul(state ^ (state >>> 15), 1 | state);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4_294_967_296;
  };
}

function buildDailyStats(): DailyStat[] {
  const random = createRandom(20_261_122);
  const today = dayjs().startOf('day');

  return Array.from({ length: DAY_COUNT }, (_, position) => {
    const daysAgo = DAY_COUNT - 1 - position;
    const date = today.subtract(daysAgo, 'day');
    const weekday = date.day();

    const growth = 10 + (position / DAY_COUNT) * 14;
    const weekdayBoost = weekday === 0 || weekday === 6 ? 1.45 : weekday === 5 ? 1.2 : 1;
    const noise = 0.55 + random() * 0.9;
    const spike = random() < 0.05 ? 1.8 : 1;
    const orders = Math.max(0, Math.round(growth * weekdayBoost * noise * spike));
    const averageOrderValue = 360 + random() * 140;

    return {
      date: date.format('YYYY-MM-DD'),
      orders,
      revenue: Math.round(orders * averageOrderValue),
    };
  });
}

export const mockDailyStats: DailyStat[] = buildDailyStats();

===== FILE: src/features/reports/components/BusiestHoursCard.tsx =====
import { Card, Tag, Typography } from 'antd';
import type { HourSlot } from '@/features/reports/types';

const BAR_AREA_HEIGHT = 220;

type BusiestHoursCardProps = {
  slots: HourSlot[];
};

/** Share of orders by time of day. The busiest slot is dark, the others light. */
export function BusiestHoursCard({ slots }: BusiestHoursCardProps) {
  const highest = Math.max(...slots.map((slot) => slot.share), 0);
  const peak = slots.find((slot) => slot.share === highest);

  return (
    <Card className="trend-card" variant="borderless">
      <div className="trend-card__head">
        <div>
          <Typography.Title className="trend-card__title" level={2}>
            Busiest hours
          </Typography.Title>
          <Typography.Text className="trend-card__subtitle">
            Share of orders by time of day, last 30 days
          </Typography.Text>
        </div>
        {peak && <Tag className="peak-tag">Peak {peak.range}</Tag>}
      </div>

      <ul className="hours-chart" style={{ height: BAR_AREA_HEIGHT + 60 }}>
        {slots.map((slot) => (
          <li className="hours-chart__slot" key={slot.range}>
            <span className="visually-hidden">{`${slot.range}: ${Math.round(slot.share * 100)}% of orders`}</span>
            <span aria-hidden="true" className="hours-chart__value">
              {Math.round(slot.share * 100)}%
            </span>
            <span
              aria-hidden="true"
              className={`hours-chart__bar${slot.share === highest ? ' hours-chart__bar--peak' : ''}`}
              style={{ height: `${highest > 0 ? (slot.share / highest) * BAR_AREA_HEIGHT : 0}px` }}
            />
            <span aria-hidden="true" className="hours-chart__label">
              {slot.label}
            </span>
          </li>
        ))}
      </ul>
    </Card>
  );
}

import { statsBetween, sumTotals } from '@/features/reports/report-utils';
import type { DailyStat } from '@/features/reports/types';
import { formatCount, formatCurrency } from '@/utils/number-formats';

const DATE_FORMAT = 'DD-MM-YYYY';

type CustomRangeCardProps = {
  stats: DailyStat[];
  today: Dayjs;
  onDownload: (start: Dayjs, end: Dayjs) => void;
};

export function CustomRangeCard({ stats, today, onDownload }: CustomRangeCardProps) {
  const [start, setStart] = useState<Dayjs | null>(null);
  const [end, setEnd] = useState<Dayjs | null>(null);
  const firstDay = dayjs(stats.at(0)?.date ?? today);

  const lastMonth = today.subtract(1, 'month');
  const presets: { label: string; start: Dayjs; end: Dayjs }[] = [
    { label: 'Last 7 days', start: today.subtract(6, 'day'), end: today },
    { label: 'Last 30 days', start: today.subtract(29, 'day'), end: today },
    { label: 'Last month', start: lastMonth.startOf('month'), end: lastMonth.endOf('month').startOf('day') },
  ];

  const changeStart = (date: Dayjs | null) => {
    setStart(date);
    // An end date before the new start date makes no sense, so it is cleared.
    if (date && end?.isBefore(date, 'day')) setEnd(null);
  };

  const range = start && end ? { start, end } : undefined;
  const totals = range ? sumTotals(statsBetween(stats, range.start, range.end)) : undefined;
  const dayCount = range ? range.end.diff(range.start, 'day') + 1 : 0;

  return (
    <Card className="range-card" variant="borderless">
      <div className="range-card__head">
        <span className="stat-tile__icon" aria-hidden="true">
          <CalendarOutlined />
        </span>
        <div>
          <Typography.Text className="stat-tile__title">Custom range</Typography.Text>
          <Typography.Text className="stat-tile__range">Pick dates or a preset</Typography.Text>
        </div>
      </div>

      <div className="range-card__presets" role="group" aria-label="Date presets">
        {presets.map((preset) => (
          <button
            aria-pressed={!!range && range.start.isSame(preset.start, 'day') && range.end.isSame(preset.end, 'day')}
            className="range-chip"
            key={preset.label}
            onClick={() => {
              setStart(preset.start);
              setEnd(preset.end);
            }}
            type="button"
          >
            {preset.label}
          </button>
        ))}
      </div>

      <div className="range-card__dates">
        <DatePicker
          aria-label="Start date"
          disabledDate={(date) => date.isAfter(end ?? today, 'day') || date.isBefore(firstDay, 'day')}
          format={DATE_FORMAT}
          onChange={changeStart}
          suffixIcon={null}
          placeholder="dd-mm-yyyy"
          value={start}
        />
        <DatePicker
          aria-label="End date"
          disabledDate={(date) => date.isAfter(today, 'day') || date.isBefore(start ?? firstDay, 'day')}
          format={DATE_FORMAT}
          onChange={setEnd}
          suffixIcon={null}
          placeholder="dd-mm-yyyy"
          value={end}
        />
      </div>

      <div className="range-card__preview" aria-live="polite">
        {totals ? (
          <>
            <strong>{formatCount(totals.orders)} orders</strong>
            <span>
              {formatCurrency(totals.revenue)} · {dayCount} {dayCount === 1 ? 'day' : 'days'}
            </span>
          </>
        ) : (
          <span>Choose a date range to see its totals.</span>
        )}
      </div>

      <Button
        block
        disabled={!range}
        icon={<DownloadOutlined />}
        onClick={() => range && onDownload(range.start, range.end)}
        type="primary"
      >
        Download report
      </Button>
    </Card>
  );
}

import { ArrowDownOutlined, ArrowUpOutlined, DownloadOutlined } from '@ant-design/icons';

const SPARK_WIDTH = 120;
const SPARK_HEIGHT = 36;

/** Tiny trend line: the recent periods, with the current one marked. */
function Sparkline({ values }: { values: number[] }) {
  if (values.length < 2) return null;
  const max = Math.max(...values, 1);
  const step = SPARK_WIDTH / (values.length - 1);
  const pointAt = (value: number, index: number) => ({
    x: index * step,
    y: SPARK_HEIGHT - 4 - (value / max) * (SPARK_HEIGHT - 8),
  });
  const points = values.map(pointAt);
  const last = points.at(-1);

  return (
    <svg className="stat-tile__spark" viewBox={`-4 0 ${SPARK_WIDTH + 8} ${SPARK_HEIGHT}`} aria-hidden="true">
      <polyline points={points.map((point) => `${point.x},${point.y}`).join(' ')} />
      {last && <circle cx={last.x} cy={last.y} r={3.5} />}
    </svg>
  );
}

function ChangeBadge({ current, previous, comparedTo }: { current: number; previous: number; comparedTo: string }) {
  const ratio = changeRatio(current, previous);
  if (ratio === undefined) {
    return <span className="stat-tile__compared">No data {comparedTo}</span>;
  }

  const direction = ratio > 0 ? 'up' : ratio < 0 ? 'down' : 'flat';
  const icon = direction === 'up' ? <ArrowUpOutlined /> : direction === 'down' ? <ArrowDownOutlined /> : null;
  const sign = direction === 'up' ? '+' : direction === 'down' ? '−' : '';

  return (
    <>
      <span className={`stat-tile__change stat-tile__change--${direction}`}>
        {icon}
        <strong>
          {sign}
          {formatPercent(Math.abs(ratio))}
        </strong>
      </span>
      <span className="stat-tile__compared">{comparedTo}</span>
    </>
  );
}

  if (range === 'month') return 'last 30 days';

type TrendCardProps = {
  stats: DailyStat[];
  today: Dayjs;
  range: TrendRange;
  onRangeChange: (range: TrendRange) => void;
  /** Shown in the footnote, e.g. 'all outlets' or 'Telibandha Raipur' */

};

/** Orders or revenue over time, with totals above the chart and a table view. */
export function TrendCard({ stats, today, range, onRangeChange, outletLabel }: TrendCardProps) {
  const [metric, setMetric] = useState<TrendMetric>('orders');
  const [view, setView] = useState<TrendView>('chart');

  const points = buildTrend(stats, range, today);
  const perUnit = range === 'year' ? 'month' : 'day';
  const chartTitle = `${metric === 'orders' ? 'Orders' : 'Revenue'} per ${perUnit}, ${describeRange(range)}`;

  const totalOrders = points.reduce((sum, point) => sum + point.orders, 0);
  const totalRevenue = points.reduce((sum, point) => sum + point.revenue, 0);
  const busiest = points.reduce<TrendPoint | undefined>(
    (best, point) => (best === undefined || point.orders > best.orders ? point : best),
    undefined,
  );
  const summary = [
    { label: 'Total orders', value: formatCount(totalOrders) },
    { label: 'Revenue', value: formatCurrency(totalRevenue) },
    {
      label: `Average per ${perUnit}`,
      value: formatCount(points.length > 0 ? Math.round(totalOrders / points.length) : 0),
    },
    {
      label: `Busiest ${perUnit}`,
      value: `${formatCount(busiest?.orders ?? 0)} orders`,
      note: busiest?.fullLabel,
    },
  ];

  return (
    <Card className="trend-card" variant="borderless">
      <div className="trend-card__head">
        <div>
          <Typography.Title className="trend-card__title" level={2}>
            Ordering trend
          </Typography.Title>
          <Typography.Text className="trend-card__subtitle">{chartTitle}</Typography.Text>
        </div>
        <div className="trend-card__controls">

            onChange={onRangeChange}
            options={rangeOptions}
            value={range}

===== FILE: src/features/reports/components/TrendChart.tsx =====
import { useState, type KeyboardEvent, type PointerEvent } from 'react';
import { formatMetric, metricValue, niceTicks } from '@/features/reports/report-utils';
import type { TrendMetric, TrendPoint } from '@/features/reports/types';
import { useElementWidth } from '@/hooks/useElementWidth';
import { formatCompactCurrency, formatCount } from '@/utils/number-formats';

const HEIGHT = 300;
const MARGIN = { top: 16, right: 20, bottom: 32, left: 56 };
/** Minimum space per x-axis label, so labels never collide. */
const MIN_LABEL_SPACING = 64;
const TOOLTIP_HALF_WIDTH = 90;

type TrendChartProps = {
  points: TrendPoint[];
  metric: TrendMetric;
  /** Accessible name, e.g. 'Orders per day, last 30 days' */
  title: string;
};

/**
 * Line + area chart for one series. Hover (or focus and use the arrow keys) to read
 * each point: a crosshair snaps to the nearest date and a tooltip shows its value.
 */
export function TrendChart({ points, metric, title }: TrendChartProps) {
  const [containerRef, width] = useElementWidth<HTMLDivElement>();
  const [activeIndex, setActiveIndex] = useState<number | null>(null);

  const values = points.map((point) => metricValue(point, metric));
  const ticks = niceTicks(Math.max(...values, 0));
  const top = ticks.at(-1) ?? 1;

  const innerWidth = Math.max(width - MARGIN.left - MARGIN.right, 0);
  const innerHeight = HEIGHT - MARGIN.top - MARGIN.bottom;
  const baseline = MARGIN.top + innerHeight;
  const lastIndex = points.length - 1;
  const peakIndex = values.indexOf(Math.max(...values));

  const x = (index: number) => MARGIN.left + (lastIndex <= 0 ? innerWidth / 2 : (index / lastIndex) * innerWidth);
  const y = (value: number) => MARGIN.top + innerHeight - (value / top) * innerHeight;

  const linePath = values.map((value, index) => `${index === 0 ? 'M' : 'L'}${x(index)},${y(value)}`).join(' ');
  const areaPath = `${linePath} L${x(lastIndex)},${baseline} L${x(0)},${baseline} Z`;

  const maxLabels = Math.max(Math.floor(innerWidth / MIN_LABEL_SPACING), 2);
  const labelEvery = Math.max(Math.ceil(points.length / maxLabels), 1);
  const showLabel = (index: number) => (lastIndex - index) % labelEvery === 0;

  const formatTick = (value: number) => (metric === 'orders' ? formatCount(value) : formatCompactCurrency(value));

  const nearestIndex = (pointerX: number) => {
    if (lastIndex <= 0) return 0;
    const ratio = (pointerX - MARGIN.left) / innerWidth;
    return Math.min(Math.max(Math.round(ratio * lastIndex), 0), lastIndex);
  };

  const handlePointerMove = (event: PointerEvent<SVGSVGElement>) => {
    const bounds = event.currentTarget.getBoundingClientRect();
    setActiveIndex(nearestIndex(event.clientX - bounds.left));
  };

  const handleKeyDown = (event: KeyboardEvent<SVGSVGElement>) => {
    const current = activeIndex ?? lastIndex;
    let next: number;
    switch (event.key) {
      case 'ArrowLeft':
        next = current - 1;
        break;
      case 'ArrowRight':
        next = current + 1;
        break;
      case 'Home':
        next = 0;
        break;
      case 'End':
        next = lastIndex;
        break;
      default:
        return;
    }
    event.preventDefault();
    setActiveIndex(Math.min(Math.max(next, 0), lastIndex));
  };

  const activePoint = activeIndex === null ? undefined : points.at(activeIndex);
  const activeValue = activeIndex === null ? undefined : values.at(activeIndex);

  return (
    <div className="trend-chart" ref={containerRef}>
      {width > 0 && points.length > 0 && (
        <svg
          className="trend-chart__svg"
          width={width}
          height={HEIGHT}
          role="img"
          aria-label={`${title}. Use the left and right arrow keys to read each value.`}
          tabIndex={0}
          onPointerMove={handlePointerMove}
          onPointerLeave={() => setActiveIndex(null)}
          onFocus={() => setActiveIndex((index) => index ?? lastIndex)}
          onBlur={() => setActiveIndex(null)}
          onKeyDown={handleKeyDown}
        >
          <defs>
            <linearGradient id="trend-area-fill" x1="0" x2="0" y1="0" y2="1">
              <stop offset="0%" className="trend-chart__area-stop-top" />
              <stop offset="100%" className="trend-chart__area-stop-bottom" />
            </linearGradient>
          </defs>

          {/* Recessive grid and y-axis labels */}
          {ticks.map((tick) => (
            <g key={tick}>
              <line
                className="trend-chart__grid"
                x1={MARGIN.left}
                x2={MARGIN.left + innerWidth}
                y1={y(tick)}
                y2={y(tick)}
              />
              <text
                className="trend-chart__axis-label"
                x={MARGIN.left - 10}
                y={y(tick)}
                textAnchor="end"
                dominantBaseline="middle"
              >
                {formatTick(tick)}
              </text>
            </g>
          ))}

          {/* X-axis labels, thinned so they never overlap */}
          {points.map((point, index) =>
            showLabel(index) ? (
              <text
                key={point.key}
                className="trend-chart__axis-label"
                x={x(index)}
                y={baseline + 20}
                textAnchor={index === 0 ? 'start' : index === lastIndex ? 'end' : 'middle'}
              >
                {point.label}
              </text>
            ) : null,
          )}

          <path className="trend-chart__area" d={areaPath} />
          <path className="trend-chart__line" d={linePath} />

          {/* The highest point is always marked */}
          {peakIndex >= 0 && (
            <circle className="trend-chart__peak" cx={x(peakIndex)} cy={y(values.at(peakIndex) ?? 0)} r={5} />
          )}

          {activeIndex !== null && activeValue !== undefined && (
            <g className="trend-chart__focus">
              <line x1={x(activeIndex)} x2={x(activeIndex)} y1={MARGIN.top} y2={baseline} />
              <circle cx={x(activeIndex)} cy={y(activeValue)} r={5} />
            </g>
          )}
        </svg>
      )}

      {activePoint && activeIndex !== null && activeValue !== undefined && (
        <div
          className="trend-chart__tooltip"
          role="status"
          style={{
            left: Math.min(Math.max(x(activeIndex), TOOLTIP_HALF_WIDTH), width - TOOLTIP_HALF_WIDTH),
            top: y(activeValue),
          }}
        >
          <span className="trend-chart__tooltip-value">
            <span className="trend-chart__tooltip-key" aria-hidden="true" />
            {formatMetric(activeValue, metric)}
          </span>

import dayjs, { type Dayjs } from 'dayjs';
import { mockOutlets, mockProblemStatusShares } from '@/features/reports/api/mock-report-extras';
import type {
  DailyStat,
  OrderStatusSlice,
  ReportPeriodKind,

  TopOutlet,
  TrendMetric,
  TrendPoint,
  TrendRange,
} from '@/features/reports/types';
import { toCsv } from '@/lib/download';
import { formatCount, formatCurrency } from '@/utils/number-formats';

export function metricValue(point: TrendPoint, metric: TrendMetric): number {
  return metric === 'orders' ? point.orders : point.revenue;
}

/** 12 → '12 orders', 5400 → '₹5,400' */
export function formatMetric(value: number, metric: TrendMetric): string {
  return metric === 'orders' ? `${formatCount(value)} ${value === 1 ? 'order' : 'orders'}` : formatCurrency(value);
}

export type Totals = { orders: number; revenue: number };

export type PeriodSummary = {
  start: Dayjs;
  end: Dayjs;
  current: Totals;
  /** Same length of time, one period earlier (yesterday, same days last week, ...) */
  previous: Totals;
  /** Up to 12 recent periods (oldest first), for the sparkline */
  history: number[];
};

const DATE_KEY = 'YYYY-MM-DD';

export function statsBetween(stats: DailyStat[], start: Dayjs, end: Dayjs): DailyStat[] {
  const from = start.format(DATE_KEY);
  const to = end.format(DATE_KEY);
  return stats.filter((stat) => stat.date >= from && stat.date <= to);
}

export function sumTotals(stats: DailyStat[]): Totals {
  return stats.reduce((sum, stat) => ({ orders: sum.orders + stat.orders, revenue: sum.revenue + stat.revenue }), {
    orders: 0,
    revenue: 0,
  });
}

/** Weeks start on Monday. */
export function startOfWeek(date: Dayjs): Dayjs {
  return date.subtract((date.day() + 6) % 7, 'day').startOf('day');
}

function totalsBetween(stats: DailyStat[], start: Dayjs, end: Dayjs): Totals {
  return sumTotals(statsBetween(stats, start, end));
}

/** Today / this week / this month so far, compared with the same stretch of the previous period. */
export function summarizePeriod(stats: DailyStat[], kind: ReportPeriodKind, today: Dayjs): PeriodSummary {
  if (kind === 'today') {
    const yesterday = today.subtract(1, 'day');
    return {
      start: today,
      end: today,
      current: totalsBetween(stats, today, today),
      previous: totalsBetween(stats, yesterday, yesterday),
      history: Array.from({ length: 12 }, (_, index) => {
        const day = today.subtract(11 - index, 'day');
        return totalsBetween(stats, day, day).orders;
      }),
    };
  }

  if (kind === 'week') {
    const start = startOfWeek(today);
    const elapsedDays = today.diff(start, 'day');
    const previousStart = start.subtract(1, 'week');
    return {
      start,
      end: today,
      current: totalsBetween(stats, start, today),
      previous: totalsBetween(stats, previousStart, previousStart.add(elapsedDays, 'day')),
      history: Array.from({ length: 12 }, (_, index) => {
        const weekStart = start.subtract(11 - index, 'week');
        return totalsBetween(stats, weekStart, weekStart.add(6, 'day')).orders;
      }),
    };
  }

  const start = today.startOf('month');
  const previousStart = start.subtract(1, 'month');
  // Same day of the month, capped at the previous month's last day (e.g. 31 Mar → 29 Feb).
  const previousEnd = previousStart.date(Math.min(today.date(), previousStart.daysInMonth()));
  return {
    start,
    end: today,
    current: totalsBetween(stats, start, today),
    previous: totalsBetween(stats, previousStart, previousEnd),
    history: Array.from({ length: 12 }, (_, index) => {
      const monthStart = start.subtract(11 - index, 'month');
      return totalsBetween(stats, monthStart, monthStart.endOf('month')).orders;
    }),
  };
}

/** Points for the trend chart: last 7 days, last 30 days, or last 12 months. */
export function buildTrend(stats: DailyStat[], range: TrendRange, today: Dayjs): TrendPoint[] {
  if (range === 'year') {
    return Array.from({ length: 12 }, (_, index) => {
      const monthStart = today.startOf('month').subtract(11 - index, 'month');
      const totals = totalsBetween(stats, monthStart, monthStart.endOf('month'));
      return {
        key: monthStart.format('YYYY-MM'),
        label: monthStart.format('MMM'),
        fullLabel: monthStart.isSame(today, 'month')
          ? `${monthStart.format('MMMM YYYY')} (so far)`
          : monthStart.format('MMMM YYYY'),
        ...totals,
      };
    });
  }

  const dayCount = range === 'week' ? 7 : 30;
  return Array.from({ length: dayCount }, (_, index) => {
    const day = today.subtract(dayCount - 1 - index, 'day');
    return {
      key: day.format(DATE_KEY),
      label: range === 'week' ? day.format('ddd') : day.format('DD MMM'),
      fullLabel: day.format('ddd, DD MMM YYYY'),
      ...totalsBetween(stats, day, day),
    };
  });
}

/** Round axis ticks (0, 10, 20, ...) that cover `max`, about five of them. */
export function niceTicks(max: number, targetCount = 5): number[] {
  if (max <= 0) return [0, 1];
  const roughStep = max / (targetCount - 1);
  const magnitude = 10 ** Math.floor(Math.log10(roughStep));
  const fraction = roughStep / magnitude;
  const niceFraction = fraction <= 1 ? 1 : fraction <= 2 ? 2 : fraction <= 2.5 ? 2.5 : fraction <= 5 ? 5 : 10;
  const step = Math.max(niceFraction * magnitude, 1);
  const top = Math.ceil(max / step) * step;
  return Array.from({ length: Math.round(top / step) + 1 }, (_, index) => index * step);
}

/** Change from previous to current as a ratio (0.12 = +12%), or undefined when there is nothing to compare. */
export function changeRatio(current: number, previous: number): number | undefined {
  if (previous === 0) return undefined;
  return (current - previous) / previous;
}

/** CSV with one row per day plus a total row. */
export function buildDailyCsv(stats: DailyStat[]): string {
  const totals = sumTotals(stats);
  const average = (revenue: number, orders: number) => (orders > 0 ? Math.round(revenue / orders) : 0);

  return toCsv([
    ['Date', 'Day', 'Orders', 'Revenue (INR)', 'Average order value (INR)'],
    ...stats.map((stat) => [
      stat.date,
      dayjs(stat.date).format('dddd'),
      stat.orders,
      stat.revenue,
      average(stat.revenue, stat.orders),
    ]),
    ['Total', '', totals.orders, totals.revenue, average(totals.revenue, totals.orders)],
  ]);
}

/** 'order-report_2026-11-01_to_2026-11-22.csv', or 'order-report_telibandha-raipur_2026-11-01_to_2026-11-22.csv' for one outlet */
export function reportFileName(start: Dayjs, end: Dayjs, outletName?: string): string {
  const from = start.format(DATE_KEY);
  const to = end.format(DATE_KEY);
  const period = from === to ? from : `${from}_to_${to}`;
  const outletPart = outletName ? `${outletName.toLowerCase().replace(/[^a-z0-9]+/g, '-')}_` : '';
  return `order-report_${outletPart}${period}.csv`;
}

/** Average order value in rupees, rounded. 0 when there are no orders. */
export function averageOrderValue({ orders, revenue }: Totals): number {
  return orders > 0 ? Math.round(revenue / orders) : 0;
}

/** The days a trend range covers: last 7 days, last 30 days, or from the first day of the month 11 months ago. */
export function trendPeriod(range: TrendRange, today: Dayjs): { start: Dayjs; end: Dayjs } {
  if (range === 'year') return { start: today.startOf('month').subtract(11, 'month'), end: today };
  return { start: today.subtract(range === 'week' ? 6 : 29, 'day'), end: today };
}

/** The numbers of one outlet: every day's orders and revenue times its share of all orders. */
export function scaleStats(stats: DailyStat[], share: number): DailyStat[] {
  if (share === 1) return stats;
  return stats.map((stat) => ({
    ...stat,
    orders: Math.round(stat.orders * share),
    revenue: Math.round(stat.revenue * share),
  }));
}

/** Splits an order count into statuses. Delivered gets what is left, so the parts always add up. */
export function buildStatusBreakdown(totalOrders: number): OrderStatusSlice[] {
  const problems = mockProblemStatusShares.map(({ key, label, share }) => ({
    key,
    label,
    count: Math.round(totalOrders * share),
  }));
  const delivered = totalOrders - problems.reduce((sum, slice) => sum + slice.count, 0);
  const slices = [{ key: 'DELIVERED' as const, label: 'Delivered', count: delivered }, ...problems];
  return slices.map((slice) => ({ ...slice, share: totalOrders > 0 ? slice.count / totalOrders : 0 }));
}

/** Orders per outlet for the top list, highest first. */
export function buildTopOutlets(totalOrders: number): TopOutlet[] {
  return mockOutlets
    .map((outlet: ReportOutlet) => ({
      id: outlet.id,
      name: outlet.name,
      orders: Math.round(totalOrders * outlet.share),
    }))
    .sort((a, b) => b.orders - a.orders);
}

/** 0.927 → '92.7%' (one decimal, so small shares stay readable) */
export function formatShare(ratio: number): string {
  return `${(ratio * 100).toFixed(1)}%`;
}
