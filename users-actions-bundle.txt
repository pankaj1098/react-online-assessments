import dayjs, { type Dayjs } from 'dayjs';
import { z } from 'zod';

/**
 * Small building blocks shared by the Zod schemas of different forms,
 * so every form words its messages and handles empty input the same way.
 */

/** Required text: trimmed, not empty, with a maximum length. */
export function requiredText(message: string, max: number) {
  return z.string({ error: message }).trim().min(1, message).max(max, `Use ${max} characters or fewer`);
}

/** Optional text: empty input becomes `undefined` in the payload. */
export function optionalText(max: number) {
  return z
    .string()
    .trim()
    .max(max, `Use ${max} characters or fewer`)
    .nullish()
    .transform((value) => value || undefined);
}

function isValidDate(value: unknown): value is Dayjs {
  return dayjs.isDayjs(value) && value.isValid();
}

/**
 * Required date and time. The form holds a dayjs value (what Ant Design's DatePicker gives us),
 * the payload gets an ISO text such as "2026-12-01T09:30:00.000Z".
 */
export function requiredDateTime(message: string) {
  return z.custom<Dayjs>(isValidDate, { error: message }).transform((value) => value.toISOString());
}

/** Optional date and time. Empty input becomes `undefined` in the payload. */
export function optionalDateTime(message: string) {
  return z
    .custom<Dayjs>(isValidDate, { error: message })
    .nullish()
    .transform((value) => value?.toISOString());
}

/**
 * Optional date that is today or earlier, such as a birthday. The form holds a dayjs value (what the DatePicker
 * gives us), the payload gets "2026-12-25". Empty input becomes `undefined`.
 */
export function optionalPastDate() {
  return z
    .custom<Dayjs>(isValidDate, { error: 'Pick a date' })
    .nullish()
    .refine((value) => !value || !value.isAfter(dayjs(), 'day'), 'The date can not be in the future')
    .transform((value) => value?.format('YYYY-MM-DD'));
}

/** Optional whole number from 0 up to `max`. Empty input becomes `undefined`. */
export function optionalWholeNumber(max = 1_000_000) {
  return z
    .number({ error: 'Enter a number' })
    .int('Use a whole number')
    .min(0, 'Use 0 or more')
    .max(max, `Use ${max} or less`)
    .nullish()
    .transform((value) => value ?? undefined);
}

/** Optional amount of money (or any decimal number) from 0 up to `max`. Empty input becomes `undefined`. */
export function optionalAmount(max = 10_000_000) {
  return z
    .number({ error: 'Enter a number' })
    .min(0, 'Use 0 or more')
    .max(max, `Use ${max} or less`)
    .nullish()
    .transform((value) => value ?? undefined);
}

/** Optional percentage from 0 to 100. Empty input becomes `undefined`. */
export function optionalPercent() {
  return z
    .number({ error: 'Enter a number' })
    .min(0, 'Use 0 or more')
    .max(100, 'Use 100 or less')
    .nullish()
    .transform((value) => value ?? undefined);
}

function isJsonObject(text: string | undefined): boolean {
  if (!text) return true;
  try {
    const parsed: unknown = JSON.parse(text);
    return typeof parsed === 'object' && parsed !== null && !Array.isArray(parsed);
  } catch {
    return false;
  }
}

/**
 * Optional JSON typed into a text box. It must be an object such as {"key": "value"}.
 * The payload gets the parsed object, or `undefined` when the box is empty.
 */
export function optionalJsonObject(maxLength = 5000) {
  return optionalText(maxLength)
    .refine(isJsonObject, 'Enter a valid JSON object, for example {"key": "value"}')
    .transform((text) => (text ? (JSON.parse(text) as Record<string, unknown>) : undefined));
}

import { DeleteOutlined, EditOutlined } from '@ant-design/icons';
import { Button, Flex, Popconfirm, Tooltip } from 'antd';
import type { ReactNode } from 'react';

type RowActionsProps = {
  /** Name of the row, used in button labels and in the delete question, for example "Chicken". */
  name: string;
  /** Short word for what the row is, for example "category". */
  what: string;
  onEdit: () => void;
  onDelete: () => void;
  /** More buttons, shown before Edit. */
  extra?: ReactNode;
  /** Text under the delete question. Change it when the row is not removed for good right away. */
  deleteDescription?: string;
};

/**
 * The Edit and Delete buttons at the end of a table row. Delete asks "are you sure?" first.
 * Hiding a button is not security: the server must check who may edit or delete.
 */
export function RowActions({ name, what, onEdit, onDelete, extra, deleteDescription }: RowActionsProps) {
  return (
    <Flex align="center" gap={4}>
      {extra}
      <Tooltip title={`Edit ${what}`}>
        <Button size="small" type="text" icon={<EditOutlined />} aria-label={`Edit ${name}`} onClick={onEdit} />
      </Tooltip>
      <Popconfirm
        title={`Delete this ${what}?`}
        description={deleteDescription ?? `"${name}" will be removed permanently.`}
        okText="Delete"
        okButtonProps={{ danger: true }}
        cancelText="Cancel"
        placement="topRight"
        onConfirm={onDelete}
      >
        <Tooltip title={`Delete ${what}`}>
          <Button size="small" type="text" danger icon={<DeleteOutlined />} aria-label={`Delete ${name}`} />
        </Tooltip>
      </Popconfirm>
    </Flex>
  );
}

/** '/orders/123' → 'orders' */
export function getModuleKey(pathname: string): string {
  return pathname.split('/')[1] ?? '';
}

/**
 * '/orders' → 'Orders', '/outlets/new' → 'Create outlet', '/apps/42/edit' → 'Edit app'.
 * Looks for a page that matches the full address first, then falls back to the module
 * ('/orders/123' → 'Orders').
 */
export function getPageTitle(pathname: string): string {
  const page = allPages.find((candidate) => matchPath(`/${candidate.key}`, pathname));
  return page?.label ?? pageTitles.get(getModuleKey(pathname)) ?? '';
}

/**
 * The users list. Uses virtual scrolling instead of page buttons: the list grows as the person
 * scrolls down (the next page is loaded near the end), and only the rows on screen are drawn.
 */
export function UsersTable({ users, loading = false, onReachEnd, ...actions }: UsersTableProps) {
  const { tableRef, bodyHeight } = useTableBodyHeight();
  // On a phone the Actions column scrolls with the table: pinned, it would cover the user's details.
  const isMobile = !Grid.useBreakpoint().md;
  const columns = getColumns(actions, isMobile);
  const tableWidth = columns.reduce((sum, column) => sum + Number(column.width ?? 0), 0);

  const handleScroll = (event: UIEvent<HTMLDivElement>) => {
    const list = event.currentTarget;
    if (list.scrollHeight - list.scrollTop - list.clientHeight < LOAD_MORE_DISTANCE) onReachEnd();
  };

  return (
    <Table<User>
      ref={tableRef}
      className="list-table users-table"
      virtual
      rowKey="id"
      columns={columns}
      dataSource={users}
      loading={loading}
      pagination={false}
      scroll={{ x: tableWidth, y: bodyHeight }}
      onScroll={handleScroll}
      locale={{ emptyText: <Empty image={Empty.PRESENTED_IMAGE_SIMPLE} description="No users found" /> }}
    />
  );
}

          <>
            <Button icon={<SyncOutlined />} onClick={() => void navigate('/users/force-update')}>
              Force update
            </Button>
            <Button icon={<EnvironmentOutlined />} onClick={() => void navigate('/users/non-serviceable')}>
              Non serviceable
            </Button>
            <Button type="primary" icon={<DatabaseOutlined />} onClick={() => void navigate('/users/tasks')}>
              Generate dump
            </Button>
          </>
        }
        footer={
          total !== undefined &&
          `Showing ${formatCount(users.length)} of ${formatCount(total)} users${
            usersQuery.hasNextPage ? ' · Scroll down to load more' : ''
          } · Sample data`
        }
      >
        <UserSearchBar onSearch={setSearch} />

        {usersQuery.isError && users.length === 0 ? (
          <LoadError what="users" onRetry={() => void usersQuery.refetch()} />
        ) : (
          <UsersTable
            users={users}
            loading={usersQuery.isPending}
            onReachEnd={loadMore}
            lockingUserId={lockMutation.isPending ? lockMutation.variables.userId : undefined}
            forceUpdatingUserId={forceUpdateMutation.isPending ? forceUpdateMutation.variables.userId : undefined}
            onEdit={(user) => void navigate(`/users/${encodeURIComponent(user.id)}/edit`)}
            onViewReferral={setReferralUser}
            onViewOrders={(user) => void navigate(`/users/${encodeURIComponent(user.id)}/orders`)}
            onViewWallet={setWalletUser}
            onToggleOtpLock={(user) => lockMutation.mutate({ userId: user.id, locked: !user.otpLoginLocked })}
            onToggleForceUpdate={(user) => forceUpdateMutation.mutate({ userId: user.id, enabled: !user.forceUpdate })}
          />
        )}
      </ListPanel>

      <ReferralModal user={referralUser} onClose={() => setReferralUser(undefined)} />
      {/* The key starts the wallet window on the newest transactions every time it is opened. */}
      <WalletTransactionsModal
        key={walletUser?.id ?? 'closed'}
        user={walletUser}
        onClose={() => setWalletUser(undefined)}
      />
    </div>
  );
}
