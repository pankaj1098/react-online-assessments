import { SearchOutlined, ShopOutlined } from '@ant-design/icons';
import { useMutation } from '@tanstack/react-query';
import { App, Card, Input, Select, Space, Tabs, Tag, Typography } from 'antd';
import { useState } from 'react';
import { useSearchParams } from 'react-router-dom';
import { mockOrders } from '@/features/orders/api/mock-orders';
import { updateOrder, type UpdateOrderInput } from '@/features/orders/api/update-order';
import { EditOrderModal } from '@/features/orders/components/EditOrderModal';
import { OrderDetailsDrawer } from '@/features/orders/components/OrderDetailsDrawer';
import { OrdersTable } from '@/features/orders/components/OrdersTable';
import { getOrderStatusMeta } from '@/features/orders/format';
import { ORDER_STATUSES, type Order, type OrderStatus } from '@/features/orders/types';
import './OrdersPage.css';

type SearchField = 'phone' | 'email' | 'orderId';
type StatusFilter = 'ALL' | OrderStatus;

type SearchFieldDetails = { label: string; placeholder: string; inputMode: 'tel' | 'email' | 'text' };

const searchFields = new Map<SearchField, SearchFieldDetails>([
  ['phone', { label: 'Phone', placeholder: 'Search by phone number', inputMode: 'tel' }],
  ['email', { label: 'Email', placeholder: 'Search by customer email', inputMode: 'email' }],
  ['orderId', { label: 'Order ID', placeholder: 'Search by order ID', inputMode: 'text' }],
]);

const searchFieldOptions = [...searchFields].map(([field, details]) => ({ value: field, label: details.label }));

function getSearchField(field: SearchField): SearchFieldDetails {
  return searchFields.get(field) ?? { label: 'Search', placeholder: 'Search', inputMode: 'text' };
}

function matchesSearch(order: Order, field: SearchField, rawSearch: string): boolean {
  const search = rawSearch.trim().toLowerCase();
  if (!search) return true;

  if (field === 'phone') {
    const digits = search.replace(/\D/g, '');
    return digits.length > 0 && order.customer.phone.includes(digits);
  }
  if (field === 'email') return (order.customer.email ?? '').toLowerCase().includes(search);
  return (order.orderId ?? '').toLowerCase().includes(search);
}

const MAX_SEARCH_LENGTH = 100;

/** Picks the search box for text that came from outside (the Dashboard search): an @ means email, digits mean phone. */
function guessSearchField(text: string): SearchField {
  if (text.includes('@')) return 'email';
  if (/^[\d\s+-]+$/.test(text)) return 'phone';
  return 'orderId';
}

export default function OrdersPage() {
  const { message } = App.useApp();
  // The Dashboard search opens this page as /orders?search=text.
  const [searchParams] = useSearchParams();
  const startingSearch = (searchParams.get('search') ?? '').trim().slice(0, MAX_SEARCH_LENGTH);
  // TODO: Replace the sample data with a TanStack Query call once the orders API is ready.
  const [orders, setOrders] = useState<Order[]>(mockOrders);
  const [outletFilter, setOutletFilter] = useState<string>();
  const [searchField, setSearchField] = useState<SearchField>(
    startingSearch ? guessSearchField(startingSearch) : 'phone',
  );
  const [search, setSearch] = useState(startingSearch);
  const [statusFilter, setStatusFilter] = useState<StatusFilter>('ALL');
  const [viewingKey, setViewingKey] = useState<string>();
  const [isDrawerOpen, setIsDrawerOpen] = useState(false);
  const [editingKey, setEditingKey] = useState<string>();
  const activeSearch = getSearchField(searchField);

  const outletOptions = [...new Set(orders.map((order) => order.outletName))]
    .sort((a, b) => a.localeCompare(b))
    .map((outlet) => ({ value: outlet, label: outlet }));

  // Outlet and search filters first, so the status tab counts reflect them.
  const filteredOrders = orders.filter(
    (order) => (!outletFilter || order.outletName === outletFilter) && matchesSearch(order, searchField, search),
  );
  const countByStatus = (status: OrderStatus) => filteredOrders.filter((order) => order.status === status).length;
  const visibleOrders =
    statusFilter === 'ALL' ? filteredOrders : filteredOrders.filter((order) => order.status === statusFilter);

  const statusTabs = [
    { key: 'ALL', label: 'All orders', count: filteredOrders.length },
    ...ORDER_STATUSES.map((status) => ({
      key: status,
      label: getOrderStatusMeta(status).label,
      count: countByStatus(status),
    })),
  ].map((tab) => ({
    key: tab.key,
    label: (
      <span className="orders-tab">
        {tab.label}
        <span className="orders-tab__count">{tab.count}</span>
      </span>
    ),
  }));

  const viewingOrder = orders.find((order) => order.key === viewingKey);
  const editingOrder = orders.find((order) => order.key === editingKey);

  const mutation = useMutation({
    mutationFn: updateOrder,
    onSuccess: (saved: UpdateOrderInput) => {
      setOrders((current) =>
        current.map((order) =>
          order.key === saved.key
            ? {
                ...order,
                status: saved.status,
                instructions: saved.instructions,
                // The detailed state belongs to the old status, so drop it when the status changes.
                subStatus: saved.status === order.status ? order.subStatus : undefined,
              }
            : order,
        ),
      );
      setEditingKey(undefined);
      void message.success(saved.status === 'REJECTED' ? 'The order was rejected.' : 'The order was updated.');
    },
    onError: () => {
      void message.error('The order could not be updated. Please try again.');
    },
  });

  const openOrder = (order: Order) => {
    setViewingKey(order.key);
    setIsDrawerOpen(true);
  };

  return (
    <div className="orders-page">
      <Card className="orders-panel" variant="borderless">
        <div className="orders-panel__head">
          <div>
            <Typography.Title className="orders-panel__title" level={4}>
              All orders
              <Tag className="orders-panel__count">{orders.length}</Tag>
            </Typography.Title>
            <Typography.Text className="orders-panel__subtitle">
              Track orders across outlets, update their status and review order details.
            </Typography.Text>
          </div>

          <div className="orders-panel__toolbar">
            <Select
              allowClear
              showSearch
              className="orders-panel__outlet"
              placeholder="All outlets"
              suffixIcon={<ShopOutlined />}
              options={outletOptions}
              value={outletFilter}
              onChange={setOutletFilter}
              aria-label="Filter by outlet"
            />
            <Space.Compact className="orders-panel__search">
              <Select
                value={searchField}
                options={searchFieldOptions}
                onChange={(field) => {
                  setSearchField(field);
                  setSearch('');
                }}
                popupMatchSelectWidth={false}
                aria-label="Search by"
              />
              <Input
                allowClear
                prefix={<SearchOutlined />}
                placeholder={activeSearch.placeholder}
                inputMode={activeSearch.inputMode}
                value={search}
                onChange={(event) => setSearch(event.target.value)}
                aria-label={activeSearch.placeholder}
              />
            </Space.Compact>
          </div>
        </div>

        <Tabs
          className="orders-panel__tabs"
          activeKey={statusFilter}
          onChange={(key) => setStatusFilter(key as StatusFilter)}
          items={statusTabs}
        />

        <OrdersTable orders={visibleOrders} onView={openOrder} onEdit={(order) => setEditingKey(order.key)} />

        <Typography.Text className="orders-panel__footer">
          Showing {visibleOrders.length} of {orders.length} orders · Sample data
        </Typography.Text>
      </Card>

      <OrderDetailsDrawer
        order={viewingOrder}
        open={isDrawerOpen}
        isRejecting={mutation.isPending && mutation.variables.status === 'REJECTED'}
        onClose={() => setIsDrawerOpen(false)}
        onReject={(order) => mutation.mutate({ key: order.key, status: 'REJECTED', instructions: order.instructions })}
        onPrint={(document) =>
          void message.info(`Printing the ${document === 'receipt' ? 'receipt' : 'label'} is not connected yet.`)
        }
      />

      {editingOrder && (
        <EditOrderModal
          order={editingOrder}
          saving={mutation.isPending}
          onCancel={() => setEditingKey(undefined)}
          onSave={(order, changes) => mutation.mutate({ key: order.key, ...changes })}
        />
      )}
    </div>
  );
}

import { Card, Tag, Typography } from 'antd';
import { useElementWidth } from '@/hooks/useElementWidth';
import { formatCount } from '@/utils/number-formats';

const DAYS = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];

const HEIGHT = 230;
const SIDE_SPACE = 20;
const TOP_SPACE = 24;
const BOTTOM_SPACE = 34;
/** The highest day sits at this share of the chart height, so the line never touches the top. */
const HEADROOM = 1.5;

type ActivityChartProps = {
  /** Orders per day, Monday to Sunday */
  weeklyOrders: number[];
};

/** Weekly order volume as a line with a soft area under it. */
export function ActivityChart({ weeklyOrders }: ActivityChartProps) {
  const [chartRef, width] = useElementWidth<HTMLDivElement>();

  const innerWidth = Math.max(width - SIDE_SPACE * 2, 0);
  const baseline = HEIGHT - BOTTOM_SPACE;
  const innerHeight = baseline - TOP_SPACE;
  const top = Math.max(...weeklyOrders, 1) * HEADROOM;
  const lastIndex = weeklyOrders.length - 1;

  const x = (index: number) => SIDE_SPACE + (lastIndex <= 0 ? innerWidth / 2 : (index / lastIndex) * innerWidth);
  const y = (value: number) => baseline - (value / top) * innerHeight;

  const linePath = weeklyOrders.map((value, index) => `${index === 0 ? 'M' : 'L'}${x(index)},${y(value)}`).join(' ');
  const areaPath = `${linePath} L${x(lastIndex)},${baseline} L${x(0)},${baseline} Z`;
  const summary = weeklyOrders.map((value, index) => `${DAYS.at(index) ?? ''} ${formatCount(value)}`).join(', ');

  return (
    <Card className="dash-card dash-activity" variant="borderless">
      <div className="dash-card__head">
        <div>
          <Typography.Title className="dash-card__title" level={2}>
            Order activity
          </Typography.Title>
          <Typography.Text className="dash-card__subtitle">Weekly order volume</Typography.Text>
        </div>
        {/* TODO: remove when the chart is fed by real data. */}
        <Tag className="dash-sample-tag">SAMPLE</Tag>
      </div>

      <div className="dash-activity__chart" ref={chartRef}>
        {width > 0 && (
          <svg width={width} height={HEIGHT} role="img" aria-label={`Orders per day: ${summary}`}>
            <path className="dash-activity__area" d={areaPath} />
            <path className="dash-activity__line" d={linePath} />
            {weeklyOrders.map((value, index) => (
              <circle className="dash-activity__point" cx={x(index)} cy={y(value)} key={DAYS.at(index)} r={4.5}>
                <title>{`${DAYS.at(index) ?? ''}: ${formatCount(value)} orders`}</title>
              </circle>
            ))}
            {DAYS.map((day, index) => (
              <text className="dash-activity__day" key={day} textAnchor="middle" x={x(index)} y={HEIGHT - 8}>
                {day}
              </text>
            ))}
          </svg>
        )}
      </div>
    </Card>
  );
}

          ))}
        </ul>
      )}

        View all orders →
      </Link>
    </Card>
  );

import { formatCount, formatPercent } from '@/utils/number-formats';
import type { ComparedValue } from '@/features/dashboard/types';

/** 2 → '2 min ago', 75 → '1 h 15 min ago' */
export function formatMinutesAgo(minutes: number): string {
  if (minutes < 60) return `${minutes} min ago`;
  const hours = Math.floor(minutes / 60);
  const rest = minutes % 60;
  return rest === 0 ? `${hours} h ago` : `${hours} h ${rest} min ago`;
}

/** 1 → '1 item', 3 → '3 items' */
export function formatItemCount(count: number): string {
  return `${formatCount(count)} ${count === 1 ? 'item' : 'items'}`;
}

export type Change = {
  /** '12%' or '1' */
  text: string;
  direction: 'up' | 'down' | 'flat';
};

/** Change shown as a percentage of the previous value: 48260 vs 43090 → '12%' up. */
export function getPercentChange({ current, previous }: ComparedValue): Change {
  if (previous === 0 || current === previous) return { text: '0%', direction: 'flat' };
  const ratio = (current - previous) / previous;
  return { text: formatPercent(Math.abs(ratio)), direction: ratio > 0 ? 'up' : 'down' };
}

/** Change shown as a plain number: 3 vs 2 → '1' up. Used for small counts such as cancelled orders. */
export function getCountChange({ current, previous }: ComparedValue): Change {
  const difference = current - previous;
  if (difference === 0) return { text: '0', direction: 'flat' };
  return { text: formatCount(Math.abs(difference)), direction: difference > 0 ? 'up' : 'down' };
}

/** 3 of 38 → '7.9%' (one decimal, so small shares stay readable) */
export function formatShare(part: number, total: number): string {
  if (total === 0) return '0%';
  return `${((part / total) * 100).toFixed(1)}%`;
}

import { keepPreviousData, useQuery } from '@tanstack/react-query';
import { Skeleton } from 'antd';
import { useState } from 'react';
import { LoadError } from '@/components/ui/LoadError';
import { dashboardQueryKeys } from '@/features/dashboard/api/query-keys';
import { getDashboardSummary, listCurrentOrders } from '@/features/dashboard/api/dashboard-api';
import { ActivityChart } from '@/features/dashboard/components/ActivityChart';
import { CurrentOrdersCard } from '@/features/dashboard/components/CurrentOrdersCard';
import { DashboardHeader } from '@/features/dashboard/components/DashboardHeader';
import { KpiCard } from '@/features/dashboard/components/KpiCard';
import { NeedsAttentionCard } from '@/features/dashboard/components/NeedsAttentionCard';
import { WorkspaceGrid } from '@/features/dashboard/components/WorkspaceGrid';
import { RANGE_OPTIONS } from '@/features/dashboard/constants';
import { formatShare, getCountChange, getPercentChange } from '@/features/dashboard/format';
import type { DashboardRange } from '@/features/dashboard/types';
import { formatCount, formatCurrency } from '@/utils/number-formats';
import './DashboardPage.css';

export default function DashboardPage() {
  const [range, setRange] = useState<DashboardRange>('today');

  // The old numbers stay on screen while the new period loads, so the page does not flash.
  const summaryQuery = useQuery({
    queryKey: dashboardQueryKeys.summary(range),
    queryFn: () => getDashboardSummary(range),
    placeholderData: keepPreviousData,
  });
  const ordersQuery = useQuery({
    queryKey: dashboardQueryKeys.currentOrders,
    queryFn: listCurrentOrders,
  });

  const summary = summaryQuery.data;
  const rangeOption = RANGE_OPTIONS.find((option) => option.value === range);
  const comparedTo = rangeOption?.comparedTo ?? '';
  const periodLabel = rangeOption?.periodLabel ?? '';

  return (
    <div className="dash-page">
      <DashboardHeader range={range} updatedAt={summary?.updatedAt} onRangeChange={setRange} />

      <div className="dash-layout">
        <div className="dash-main">
          {summaryQuery.isError ? (
            <LoadError what="dashboard numbers" onRetry={() => void summaryQuery.refetch()} />
          ) : summary ? (
            <>
              <div className="dash-kpis">
                <KpiCard
                  label={`Revenue ${periodLabel}`}
                  value={formatCurrency(summary.revenue.current)}
                  change={getPercentChange(summary.revenue)}
                  higherIsBetter
                  comparedTo={comparedTo}
                />
                <KpiCard
                  label={`Orders ${periodLabel}`}
                  value={formatCount(summary.orders.current)}
                  change={getPercentChange(summary.orders)}
                  higherIsBetter
                  comparedTo={comparedTo}
                />
                <KpiCard
                  label="Average order value"
                  value={formatCurrency(summary.averageOrderValue.current)}
                  change={getPercentChange(summary.averageOrderValue)}
                  higherIsBetter
                  comparedTo={comparedTo}
                />
                <KpiCard
                  label="Cancelled orders"
                  value={formatCount(summary.cancelledOrders.current)}
                  valueNote={`· ${formatShare(summary.cancelledOrders.current, summary.orders.current)}`}
                  change={getCountChange(summary.cancelledOrders)}
                  higherIsBetter={false}
                  comparedTo={comparedTo}
                />
              </div>

              <div className="dash-insights">
                <ActivityChart weeklyOrders={summary.weeklyOrders} />
                <NeedsAttentionCard attention={summary.attention} />
              </div>

              <WorkspaceGrid workspace={summary.workspace} />
            </>
          ) : (
            <Skeleton active paragraph={{ rows: 10 }} />
          )}
        </div>

        <CurrentOrdersCard
          orders={ordersQuery.data}
          isLoading={ordersQuery.isPending}
          isError={ordersQuery.isError}
          onRetry={() => void ordersQuery.refetch()}
        />
      </div>
    </div>
  );
}

export type DashboardRange = 'today' | '7days' | '30days';

/** A number for the chosen period and for the period before it (used for the "vs yesterday" change). */
export type ComparedValue = {
  current: number;
  previous: number;
};

export type DashboardSummary = {
  /** ISO date-time of the last refresh */
  updatedAt: string;
  revenue: ComparedValue;
  orders: ComparedValue;
  averageOrderValue: ComparedValue;
  cancelledOrders: ComparedValue;
  /** Orders per day, Monday to Sunday */
  weeklyOrders: number[];
  attention: {
    ordersWithoutOperator: number;
    ordersDelayed: number;
    failedPayments: number;
    outletsOffline: number;
    couponsExpiring: number;
  };
  workspace: {
    outlets: number;
    outletsOpen: number;
    operators: number;
    operatorsOnline: number;
    users: number;
    usersThisWeek: number;
    coupons: number;
    couponsActive: number;
    orders: number;
    catalogs: number;
    applications: number;
    /** Missing until the wallet bonus API exists */
    walletBonusCredits?: number;
  };
};

/** The stages shown on the dashboard. The backend order statuses map onto these. */

  itemCount: number;

  stage: CurrentOrderStage;
  minutesAgo: number;

  hasNoOperator?: boolean;

};
